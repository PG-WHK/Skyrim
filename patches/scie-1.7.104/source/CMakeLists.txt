cmake_minimum_required(VERSION 3.21)
project(SCIE_Compatibility LANGUAGES C CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set(ENABLE_SKYRIM_AE ON CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_SE OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_VR OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(SKSE_SUPPORT_PATCH_SAFETY ON CACHE BOOL "" FORCE)
set(FETCHCONTENT_SOURCE_DIR_HDE64 "${CMAKE_CURRENT_SOURCE_DIR}/MinHook" CACHE PATH "" FORCE)
find_package(spdlog CONFIG REQUIRED)
find_package(directxtk CONFIG REQUIRED)
add_subdirectory(CommonLibSSE-NG commonlib)
add_subdirectory(MinHook minhook-build)
file(GLOB_RECURSE SCIE_SOURCES CONFIGURE_DEPENDS "SCIE/src/*.cpp")
add_library(CraftingInventoryExtender SHARED ${SCIE_SOURCES})
target_compile_features(CraftingInventoryExtender PRIVATE cxx_std_23)
target_include_directories(CraftingInventoryExtender PRIVATE SCIE/include)
target_compile_definitions(CraftingInventoryExtender PRIVATE ENABLE_SKYRIM_AE=1 SKSE_SUPPORT_PATCH_SAFETY=1 WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE SKSEAPI=__cdecl)
target_compile_options(CraftingInventoryExtender PRIVATE /utf-8 /bigobj /EHsc /Zc:preprocessor /Zc:__cplusplus)
target_precompile_headers(CraftingInventoryExtender PRIVATE SCIE/include/PCH.h)
target_link_libraries(CraftingInventoryExtender PRIVATE CommonLibSSE::CommonLibSSE minhook spdlog::spdlog Version Dbghelp bcrypt d3d11 d3dcompiler dxgi)
if(TARGET Microsoft::DirectXTK)
  target_link_libraries(CraftingInventoryExtender PRIVATE Microsoft::DirectXTK)
endif()
